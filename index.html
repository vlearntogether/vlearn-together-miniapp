============================================================ # GROUP PAYMENT MESSAGE # ============================================================ async def group_message( update: Update, context: ContextTypes.DEFAULT_TYPE, ): if not update.message: return if not update.message.text: return message = update.message.text.strip() print( "\n========== GROUP MESSAGE ==========" ) print( message ) manual_apv = extract_apv( message ) manual_amount = extract_amount( message ) manual_date = extract_payment_date( message ) transaction_id = extract_transaction_id( message ) looks_like_aba = ( "paid by" in message.lower() or "ត្រូវបានបង់ដោយ" in message or "APV" in message.upper() or "ABA" in message.upper() ) if not looks_like_aba: return if not manual_apv: print( "No APV detected." ) return if not manual_amount: print( "No amount detected." ) return if not manual_date: print( "No payment date detected." ) return ( payment_day, payment_month, payment_year, payment_date, ) = manual_date payer_name, payer_account = extract_payer( message ) payment_id = save_aba_payment( amount=manual_amount, payment_date=payment_date, payment_day=payment_day, payment_month=payment_month, payment_year=payment_year, apv=manual_apv, transaction_id=transaction_id, payer_name=payer_name, payer_account=payer_account, source="telegram_group", ) if payment_id is None: print( "APV already exists:", manual_apv, ) return print( "ABA PAYMENT SAVED" ) print( "Amount:", manual_amount, ) print( "Date:", payment_date, ) print( "APV:", manual_apv, ) await try_match_payment( update=update, context=context, payment_id=payment_id, ) # ============================================================ # ADMIN COMMAND: /channels # ============================================================ async def channels_command( update: Update, context: ContextTypes.DEFAULT_TYPE, ): connection = sqlite3.connect( DATABASE ) connection.row_factory = sqlite3.Row cursor = connection.cursor() cursor.execute( """ SELECT * FROM course_channels ORDER BY course """ ) rows = cursor.fetchall() connection.close() if not rows: await update.message.reply_text( "No course channels detected yet.\n\n" "Post a test message in your " "private course channels." ) return text = "📚 REGISTERED COURSE CHANNELS\n\n" for row in rows: text += ( f"Course:\n" f"{row['course']}\n\n" f"Channel:\n" f"{row['channel_title']}\n\n" f"Channel ID:\n" f"{row['channel_id']}\n\n" "----------------------\n\n" ) await update.message.reply_text( text ) #
